# OGC 2026 — 2주차: ALNS 코어 (파괴 6 + 복구 3, RRT 수용, 증분 평가, NFP Minkowski 가속) 구현 계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 1주차 구성해(multi-start construct) 위에서 300초 예산 기준 Z1(tardiness)을 20% 이상 줄이는 ALNS 개선 루프를 `shell.improve()` 훅에 연결하고, 전 40개 인스턴스 feasible을 유지한다.

**Architecture:** ALNS는 순수 Python(`solver/alns/`)으로 구현하되 모든 기하 질의는 1주차 `ogc_core.Core` API(`place/remove/earliest_feasible/free_positions`)에 배치 호출한다. 해 상태는 "placements 리스트가 유일한 정본"이며, `SolutionState`가 Core 반영 + 증분 목적함수(Z1 델타, Z2/Z3 O(n) 재계산) + snapshot/restore(파괴-복구 실패·거부 시 롤백)를 관장한다. 성능 전제인 NFP 비트맵 생성은 오프셋 전수 스캔(v0)을 Clipper2 MinkowskiSum 기반(v1)으로 교체하되 **v0과 비트 단위 동일**을 테스트로 강제한다.

**Tech Stack:** Python 3.12, C++20 + Clipper2(vendored) + pybind11 (1주차 빌드 파이프라인 그대로), pytest, `baseline/utils.py`(최종 oracle).

```python
import json, random, time
import ogc_core

def _pairs(core, rng, n):
    out = []
    for _ in range(n):
        bi, bj = rng.randrange(core.n_blocks), rng.randrange(core.n_blocks)
        oi = rng.randrange(core.num_orients(bi))
        oj = rng.randrange(core.num_orients(bj))
        out.append((bi, oi, bj, oj))
    return out

def test_minkowski_bitmap_bit_identical(all_probs):
    rng = random.Random(11)
    for name in ("prob_1", "prob_3", "prob_21", "prob_23"):
        core = ogc_core.Core(json.dumps(all_probs[name]))
        for (bi, oi, bj, oj) in _pairs(core, rng, 40):
            for kind in ("static", "entry"):
                v0 = core._nfp_bitmap(bi, oi, bj, oj, kind, "scan")
                v1 = core._nfp_bitmap(bi, oi, bj, oj, kind, "mink")
                assert v0 == v1, f"{name} ({bi},{oi})x({bj},{oj}) {kind}: mink != scan"

def test_minkowski_is_faster(all_probs):
    core = ogc_core.Core(json.dumps(all_probs["prob_1"]))
    pairs = _pairs(core, random.Random(5), 100)
    def bench(method):
        t0 = time.perf_counter()
        for (bi, oi, bj, oj) in pairs:
            core._nfp_bitmap(bi, oi, bj, oj, "entry", method)   # 캐시 우회 → 매회 실측
        return (time.perf_counter() - t0) / len(pairs)

    print(f"per-pair: scan={scan_s*1e6:.0f}us mink={mink_s*1e6:.0f}us")
```

- [ ] **Step 3: C++ 구현** — `core/src/nfp.cpp`:

기존 v0의 "윈도 전 오프셋 × overlap_pos_area" 채움 루프를 `fill_bitmap_scan`으로 함수 분리하고(윈도 계산 코드는 공용 헬퍼 `bitmap_window(...)`로 추출 — v0/v1이 **동일 윈도**를 쓰는 것이 비트 동일성의 전제), 다음을 추가:

```cpp
#include "clipper2/clipper.h"
#include <cstdlib>
using namespace Clipper2Lib;

// t ∈ interior(A ⊕ (−B))  ⇔  int(A) ∩ int(B+t) ≠ ∅  ⇔  겹침 면적 > 0 (v0와 동일 의미론).
// 좌표는 SCALE 정수, 오프셋은 SCALE의 배수 → 모든 판정이 정수 정확.
static Paths64 mink_collision_region(const Path64& a, const Path64& b) {
    Path64 nb; nb.reserve(b.size());
    for (const auto& pt : b) nb.push_back(Point64(-pt.x, -pt.y));
    Paths64 sum = MinkowskiSum(a, nb, /*isClosed=*/true);
    return Union(sum, FillRule::NonZero);   // 방향 정규화(양면적=외곽, 음면적=구멍)
}

// 경계 위(IsOn) = 접촉 = 비충돌(v0의 area>0 의미론과 일치). 구멍은 winding으로 제외.
static bool point_strictly_inside(const Paths64& region, const Point64& p) {
    int wind = 0;
    for (const auto& path : region) {
        auto r = PointInPolygon(p, path);
        if (r == PointInPolygonResult::IsOn) return false;
        if (r == PointInPolygonResult::IsInside) wind += (Area(path) > 0) ? 1 : -1;
    }
    return wind > 0;
}

// layers_a/layers_b: 두 shape의 스케일 정수 레이어 다각형 (1주차 ShapeTable에 이미 존재).
// combos: kind에 따른 레이어 조합 목록 (static: k×k, entry: k×j (j≥k)) — v0의 조합 생성 코드를
//         함수로 추출해 scan/mink가 공유. bm은 bitmap_window()가 만든 빈 비트맵 (v0와 동일 윈도).
void fill_bitmap_mink(OffsetBitmap& bm,
                      const std::vector<Path64>& layers_a,
                      const std::vector<Path64>& layers_b,
                      const std::vector<std::pair<int,int>>& combos) {
    for (const auto& [ka, kb] : combos) {
        Paths64 region = mink_collision_region(layers_a[ka], layers_b[kb]);
        if (region.empty()) continue;
        for (int iy = 0; iy < bm.h; ++iy)
            for (int ix = 0; ix < bm.w; ++ix) {
                if (bm.get(ix, iy)) continue;  // 이미 hit이면 스킵
                Point64 p(int64_t(bm.dx_lo + ix) * SCALE, int64_t(bm.dy_lo + iy) * SCALE);
                if (point_strictly_inside(region, p)) bm.set(ix, iy);
            }
    }
}

static bool use_scan_method() {
    const char* m = std::getenv("OGC_NFP_METHOD");
    return m && std::string(m) == "scan";
}
```

v0 채움 루프도 같은 꼴의 `fill_bitmap_scan(bm, layers_a, layers_b, combos)`로 분리(내용은 기존 "오프셋별 overlap_pos_area OR" 그대로). `static_bm`/`entry_bm`의 lazy 빌드 지점은 `bitmap_window()`로 빈 비트맵을 만든 뒤 `use_scan_method() ? fill_bitmap_scan(...) : fill_bitmap_mink(...)`를 호출. `OffsetBitmap`에 `get(ix,iy)/set(ix,iy)`가 없다면 이 Task에서 추가(비트 인덱스 `iy*w+ix`).

`core/src/bindings.cpp`의 Core 클래스 바인딩에 테스트 전용 메서드 추가 (**캐시 우회** — `static_bm/entry_bm`을 거치지 않고 매번 새로 빌드):

```cpp
.def("_nfp_bitmap", [](Core& c, int bi, int oi, int bj, int oj,
                       const std::string& kind, const std::string& method) {
    OffsetBitmap bm = c.build_bitmap_fresh(bi, oi, bj, oj,
                                           /*entry_kind=*/kind == "entry",
                                           /*use_mink=*/method != "scan");
    return py::make_tuple(bm.dx_lo, bm.dy_lo, bm.w, bm.h,
                          py::bytes(reinterpret_cast<const char*>(bm.bits.data()),
                                    bm.bits.size() * sizeof(uint64_t)));
})
```

`Core::build_bitmap_fresh(bi, oi, bj, oj, entry_kind, use_mink)`는 nfp.cpp에 함께 추가: shape id 조회 → `bitmap_window` → 조합 목록 생성 → `fill_bitmap_scan` 또는 `fill_bitmap_mink` 호출만 하는 5줄짜리 조립 함수(캐시 삽입 없음).

```python
"""해 상태 관리: placements 리스트가 유일한 정본. Core 반영 + 증분 목적함수 + 롤백."""
import math

Placement = tuple  # (block, bay, x, y, orient, entry, exit)


def _bbox(shape):
    pts = [p for layer in shape["layers"] for p in layer]
    xs = [p[0] for p in pts]; ys = [p[1] for p in pts]
    return (min(xs), min(ys), max(xs), max(ys))


class OpContext:
    """prob_info에서 파생되는 불변 테이블. Core에 없는 데이터(workload, pref, bbox)의 유일 출처."""

    def __init__(self, prob):
        self.prob = prob
        w = prob.get("weights", {})
        self.w1 = float(w.get("w1", 1.0))
        self.w2 = float(w.get("w2", 1.0))
        self.w3 = float(w.get("w3", 1.0))
        bl = prob["blocks"]
        self.n_blocks = len(bl)
        self.release = [b["release_time"] for b in bl]
        self.due = [b["due_date"] for b in bl]
        self.proc = [b["processing_time"] for b in bl]
        self.workload = [float(b["workload"]) for b in bl]
        self.pref = [b["bay_preferences"] for b in bl]
        self.smax = [max(b["bay_preferences"]) for b in bl]
        bays = prob["bays"]
        self.n_bays = len(bays)
        self.bay_w = [b["width"] for b in bays]
        self.bay_h = [b["height"] for b in bays]
        areas = [b["width"] * b["height"] for b in bays]
        avg = sum(areas) / len(areas)
        self.u = [avg / a for a in areas]
        self.horizon = max(self.due)                 # 시간 정규화 기준 H
        self.horizon_search = max(self.due) * 3      # 삽입 탐색 상한 (1주차 decoder와 동일)
        self.bbox = [[_bbox(s) for s in b["shape"]] for b in bl]           # [blk][orient]
        self.center = [[((mn_x + mx_x) / 2.0, (mn_y + mx_y) / 2.0)
                        for (mn_x, mn_y, mx_x, mx_y) in per_blk]
                       for per_blk in self.bbox]                            # [blk][orient]
        self.diag = max((w_ * w_ + h_ * h_) ** 0.5
                        for w_, h_ in zip(self.bay_w, self.bay_h))


class SolutionState:
    def __init__(self, core, ctx, placements):
        self.core, self.ctx = core, ctx
        core.clear()
        n = core.load_placements(list(placements))
        assert n == len(placements), f"incumbent 적재 실패: {n}/{len(placements)}"
        self.placed = {p[0]: tuple(p) for p in placements}
        self._z1 = sum(max(0, p[6] - ctx.due[p[0]]) for p in placements)
        self._loads = [0.0] * ctx.n_bays
        for p in placements:
            self._loads[p[1]] += ctx.workload[p[0]]
        self._z3 = sum(ctx.smax[p[0]] - ctx.pref[p[0]][p[1]] for p in placements)

    # -- 증분 장부 ---------------------------------------------------------
    def remove_block(self, blk):
        p = self.placed.pop(blk)
        self.core.remove(blk)
        c = self.ctx
        self._z1 -= max(0, p[6] - c.due[blk])
        self._loads[p[1]] -= c.workload[blk]
        self._z3 -= c.smax[blk] - c.pref[blk][p[1]]
        return p

    def place_block(self, p):
        if not self.core.place(*p):
            return False
        blk = p[0]
        self.placed[blk] = tuple(p)
        c = self.ctx
        self._z1 += max(0, p[6] - c.due[blk])
        self._loads[p[1]] += c.workload[blk]
        self._z3 += c.smax[blk] - c.pref[blk][p[1]]
        return True

    # -- 목적함수 (utils 수식 그대로) ---------------------------------------
    def _z2_of(self, loads):
        if self.ctx.n_bays < 2:
            return 0.0
        vals = [self.ctx.u[j] * loads[j] for j in range(self.ctx.n_bays)]
        return float(math.floor(max(vals) - min(vals)))   # max pair |a-b| == max-min

    def z2(self):
        return self._z2_of(self._loads)

    def z2_with(self, bay, dw):
        loads = list(self._loads)
        loads[bay] += dw
        return self._z2_of(loads)

    def z(self):
        return (float(self._z1), self.z2(), float(self._z3))

    def objective(self):
        z1, z2, z3 = self.z()
        return self.ctx.w1 * z1 + self.ctx.w2 * z2 + self.ctx.w3 * z3

    # -- 롤백 ---------------------------------------------------------------
    def snapshot(self):
        return dict(self.placed)

    def restore(self, snap):
        # feasible 집합의 부분집합은 feasible(전 제약이 unary/pairwise) → 재배치는 순서 무관 성공
        for blk in list(self.placed):
            if self.placed[blk] != snap.get(blk):
                self.remove_block(blk)
        for blk, p in snap.items():
            if blk not in self.placed:
                ok = self.place_block(p)
                assert ok, f"restore 실패: block {blk} 재배치 거부"

    def placements(self):
        return [self.placed[b] for b in sorted(self.placed)]
```

```python
"""파괴 연산자 6종 (설계서 §4 L3 표). 모두 (state, q, rng, ctx) -> list[removed block id]."""


def destroy_random(state, q, rng, ctx):
    ids = sorted(state.placed)
    out = rng.sample(ids, min(q, len(ids)))
    for b in out:
        state.remove_block(b)
    return out


def destroy_worst_tardiness(state, q, rng, ctx, p=3):
    """지각 기여 상위 블록을 rand^p 편향으로 제거. 지각 블록 부족분은 무작위 보충."""
    scored = sorted(((max(0, pl[6] - ctx.due[b]), b) for b, pl in state.placed.items()),
                    reverse=True)
    tardy = [b for t, b in scored if t > 0]
    rest = [b for t, b in scored if t == 0]
    out = []
    while tardy and len(out) < q:
        i = min(int(rng.random() ** p * len(tardy)), len(tardy) - 1)
        out.append(tardy.pop(i))
    while rest and len(out) < q:
        out.append(rest.pop(rng.randrange(len(rest))))
    for b in out:
        state.remove_block(b)
    return out


def relatedness(ctx, pi, pj, A, B, C):
    """Shaw 관련도: 작을수록 관련 깊음. A·d + B·|Δentry|/H + C·|Δexit|/H."""
    if pi[1] != pj[1]:
        d = 1.0                                   # 다른 bay = 공간 최대 거리로 간주
    else:
        ci = ctx.center[pi[0]][pi[4]]
        cj = ctx.center[pj[0]][pj[4]]
        dx = (pi[2] + ci[0]) - (pj[2] + cj[0])
        dy = (pi[3] + ci[1]) - (pj[3] + cj[1])
        d = (dx * dx + dy * dy) ** 0.5 / ctx.diag
    H = float(ctx.horizon)
    return A * d + B * abs(pi[5] - pj[5]) / H + C * abs(pi[6] - pj[6]) / H


def destroy_shaw(state, q, rng, ctx, A=0.55, B=1.36, C=0.89, p=6):
    ids = sorted(state.placed)
    snaps = {b: state.placed[b] for b in ids}     # 제거 전 위치 보존 (관련도 계산용)
    seed = ids[rng.randrange(len(ids))]
    removed = [seed]
    state.remove_block(seed)
    while len(removed) < min(q, len(ids)):
        r = removed[rng.randrange(len(removed))]
        rest = [b for b in ids if b in state.placed]
        rest.sort(key=lambda b: relatedness(ctx, snaps[r], snaps[b], A, B, C))
        pick = rest[min(int(rng.random() ** p * len(rest)), len(rest) - 1)]
        removed.append(pick)
        state.remove_block(pick)
    return removed
```

```python
def destroy_blocking_set(state, q, rng, ctx, max_probe=10):
    """지각 블록의 '조기 진입(entry ≤ due−P)'을 막는 방해 집합을 제거-탐침으로 역추적 제거.
    공간 선점자든 스윕 차단자든 '그 블록을 빼면 earliest_feasible이 당겨진다'로 동일하게 검출됨."""
    core = state.core

    def best_entry(blk, target):
        best = _BIG
        for bay in range(ctx.n_bays):
            for o in range(core.num_orients(blk)):
                hit = core.earliest_feasible(blk, bay, o, ctx.release[blk], target)
                if hit is not None:
                    best = min(best, hit[0])
        return best                                  # _BIG이면 target까지 진입 불가

    scored = sorted(((max(0, p[6] - ctx.due[b]), b) for b, p in state.placed.items()),
                    reverse=True)
    tardy = [b for t, b in scored if t > 0]
    if not tardy:
        return destroy_random(state, q, rng, ctx)

    removed = []
    for blk in tardy:
        if len(removed) >= q or blk not in state.placed:
            continue
        target = ctx.due[blk] - ctx.proc[blk]
        if target < ctx.release[blk]:
            continue                                 # zero-slack 이하: 조기 진입 자체가 불가
        p_blk = state.remove_block(blk)
        removed.append(blk)
        probes = 0
        while best_entry(blk, target) >= _BIG and len(removed) < q and probes < max_probe:
            cands = [p for p in state.placed.values()
                     if p[1] == p_blk[1]
                     and p[5] < target + ctx.proc[blk] and p[6] > ctx.release[blk]]
            if not cands:
                break
            cands.sort(key=lambda p: abs(p[2] - p_blk[2]) + abs(p[3] - p_blk[3]))
            base = best_entry(blk, target)
            best_gain, best_c = 0, None
            for p in cands[:max_probe]:
                pc = state.remove_block(p[0])
                e2 = best_entry(blk, target)
                assert state.place_block(pc)         # 원위치 복귀는 항상 성공(부분집합 feasible)
                probes += 1
                if base - e2 > best_gain:
                    best_gain, best_c = base - e2, p[0]
            victim = best_c if best_c is not None else cands[0][0]
            state.remove_block(victim)
            removed.append(victim)
    # q 미달 시 무작위 보충 (repair에 일감 보장)
    ids = sorted(state.placed)
    while ids and len(removed) < q:
        b = ids.pop(rng.randrange(len(ids)))
        state.remove_block(b)
        removed.append(b)
    return removed
```

```python
"""복구 연산자. 후보 생성은 core.earliest_feasible/free_positions만 사용 (자체 기하 금지).
순서 결정은 1주차 priority.atc_order를 재사용한다."""
from solver.priority import atc_order


def insertion_cost(state, blk, bay, entry, ctx):
    """블록을 (bay, entry)에 넣을 때의 가중 목적 증가분: w1·지각 + w2·ΔZ2 + w3·pref 손실."""
    tard = max(0, entry + ctx.proc[blk] - ctx.due[blk])
    dz2 = state.z2_with(bay, ctx.workload[blk]) - state.z2()
    pref_loss = ctx.smax[blk] - ctx.pref[blk][bay]
    return ctx.w1 * tard + ctx.w2 * dz2 + ctx.w3 * pref_loss


def candidate_for_bay(state, blk, bay, ctx):
    """해당 bay에서의 최선 후보. bay 안에서는 비용이 entry에 단조 → 방향별 earliest의 min."""
    core = state.core
    best = None
    for o in range(core.num_orients(blk)):
        hit = core.earliest_feasible(blk, bay, o, ctx.release[blk], ctx.horizon_search)
        if hit is None:
            continue
        e, x, y = hit
        cand = (insertion_cost(state, blk, bay, e, ctx), bay, x, y, o, e)
        if best is None or cand < best:
            best = cand
    return best


def candidates_by_bay(state, blk, ctx):
    out = {}
    for bay in range(ctx.n_bays):
        c = candidate_for_bay(state, blk, bay, ctx)
        if c is not None:
            out[bay] = c
    return out


def _place_candidate(state, blk, cand, ctx):
    cost, bay, x, y, o, e = cand
    ok = state.place_block((blk, bay, x, y, o, e, e + ctx.proc[blk]))
    assert ok, f"조회 직후 배치 실패: block {blk} bay {bay} (상태 불변식 위반)"


def repair_greedy_noise(state, removed, rng, ctx, noise=0.1):
    """ATC biased-randomization 순서(디코더와 동일 기계장치) + 비용 곱셈 노이즈 U[1−η,1+η]."""
    rset = set(removed)
    order = [b for b in atc_order(ctx.prob, rng=rng) if b in rset]
    for blk in order:
        cands = candidates_by_bay(state, blk, ctx)
        if not cands:
            return False                     # 호출자(loop)가 snapshot으로 롤백
        best = min(cands.values(),
                   key=lambda c: (c[0] * (1.0 + rng.uniform(-noise, noise)), c[5], c[3], c[2]))
        _place_candidate(state, blk, best, ctx)
    return True
```

```python
def _regret_value(costs, k, M):
    """오름차순 비용 리스트의 regret-k. 대안 부족분은 M 가산으로 채운다."""
    costs = sorted(costs)
    while len(costs) < k:
        costs.append(costs[-1] + M)
    return sum(costs[l] - costs[0] for l in range(1, k))


def repair_regret(state, removed, rng, ctx, k=2):
    """regret-k 삽입. 기하 재탐색은 '방금 삽입된 bay'만, 비용 산술은 매회 전 bay 갱신.
    (배치 제약은 bay 내부 pairwise뿐이라 다른 bay 후보의 기하는 유효하게 유지됨.
     단 Z2 항은 전역이므로 비용만 재계산한다.)"""
    pending = set(removed)
    cand = {b: candidates_by_bay(state, b, ctx) for b in pending}
    M = ctx.w1 * float(ctx.horizon)
    while pending:
        best_blk, best_key = None, None
        for b in sorted(pending):
            if not cand[b]:
                return False
            regret = _regret_value([c[0] for c in cand[b].values()], k, M)
            key = (regret, -min(c[0] for c in cand[b].values()))   # 큰 regret, tie→작은 비용
            if best_key is None or key > best_key:
                best_key, best_blk = key, b
        chosen = min(cand[best_blk].values())
        _place_candidate(state, best_blk, chosen, ctx)
        pending.discard(best_blk)
        jbay = chosen[1]
        for b in pending:
            hit = candidate_for_bay(state, b, jbay, ctx)           # 기하 재탐색: jbay만
            if hit is None:
                cand[b].pop(jbay, None)
            else:
                cand[b][jbay] = hit
            for bay, c in list(cand[b].items()):                   # 비용 산술 갱신: 전 bay
                if bay == jbay:
                    continue
                cand[b][bay] = (insertion_cost(state, b, bay, c[5], ctx),
                                c[1], c[2], c[3], c[4], c[5])
    return True


def repair_most_constrained(state, removed, rng, ctx):
    """선택지 적은 블록부터: (지각 0 슬롯 수, 잔여 slack, −면적) 오름차순으로 최선 삽입."""
    core = state.core

    def zero_tard_options(blk):
        """지각 0으로 넣을 수 있는 (bay, orient) 슬롯 점수. 자유 위치가 넉넉한 슬롯은 2점,
        빡빡한 슬롯(자유 셀 ≤ 8)은 1점 — 점수 낮은 블록이 더 구속됨 → 먼저 삽입."""
        latest = ctx.due[blk] - ctx.proc[blk]
        if latest < ctx.release[blk]:
            return 0
        n = 0
        for bay in range(ctx.n_bays):
            for o in range(core.num_orients(blk)):
                hit = core.earliest_feasible(blk, bay, o, ctx.release[blk], latest)
                if hit is not None:
                    e, x, y = hit
                    free = len(core.free_positions(blk, bay, o, e, e + ctx.proc[blk]))
                    n += 2 if free > 8 else 1
        return n

    def area(blk):
        mn_x, mn_y, mx_x, mx_y = ctx.bbox[blk][0]
        return (mx_x - mn_x) * (mx_y - mn_y)

    order = sorted(removed,
                   key=lambda b: (zero_tard_options(b),
                                  ctx.due[b] - ctx.proc[b] - ctx.release[b],
                                  -area(b)))
    for blk in order:
        cands = candidates_by_bay(state, blk, ctx)
        if not cands:
            return False
        _place_candidate(state, blk, min(cands.values()), ctx)
    return True
```

```python
"""수용 기준. RRT 기본(설계서 §4 L3: 초기 임계 2~5% 선형 감소), reheated SA 대안."""
import math


class RecordToRecord:
    def __init__(self, initial_obj, start_pct=0.03):
        self.thr0 = start_pct * initial_obj

    def accept(self, cand_obj, cur_obj, best_obj, progress, rng):
        thr = self.thr0 * (1.0 - min(max(progress, 0.0), 1.0))
        return cand_obj < best_obj + thr


class ReheatedSA:
    def __init__(self, initial_obj, start_pct=0.05, alpha=0.9995,
                 reheat_pct=0.5, min_ratio=1e-3):
        self.T0 = max(1e-9, start_pct * initial_obj)
        self.T = self.T0
        self.alpha = alpha
        self.reheat_pct = reheat_pct
        self.min_ratio = min_ratio
        self.reheats = 0

    def accept(self, cand_obj, cur_obj, best_obj, progress, rng):
        if cand_obj < cur_obj:
            ok = True
        else:
            ok = rng.random() < math.exp(-(cand_obj - cur_obj) / max(self.T, 1e-9))
        self.T *= self.alpha
        if self.T < self.T0 * self.min_ratio:          # 재가열: 매회 절반 높이로
            self.reheats += 1
            self.T = self.T0 * (self.reheat_pct ** self.reheats)
        return ok


def make_acceptance(cfg, initial_obj):
    if getattr(cfg, "acceptance", "rrt") == "sa":
        return ReheatedSA(initial_obj, start_pct=cfg.sa_start_pct)
    return RecordToRecord(initial_obj, start_pct=cfg.rrt_start_pct)
```

```python
"""연산자 선택. 기본은 정적 균등(설계서: 적응층 기여 ~0.14%라 연산자 품질에 집중),
--adaptive 시 세그먼트 100회·점수 33/13/9·반응 0.1의 룰렛휠."""


class OperatorSelector:
    def __init__(self, names, adaptive=False, segment=100, scores=(33, 13, 9), reaction=0.1):
        if not names:
            raise ValueError("OperatorSelector: 연산자 목록이 비었음 (disable_ops 과다?)")
        self.names = list(names)
        self.adaptive = adaptive
        self.segment = segment
        self.sigma = {"best": scores[0], "better": scores[1], "accepted": scores[2]}
        self.reaction = reaction
        self.w = {n: 1.0 for n in self.names}
        self._scores = {n: 0.0 for n in self.names}
        self._uses = {n: 0 for n in self.names}
        self._iter = 0
        self.total_uses = {n: 0 for n in self.names}

    def select(self, rng):
        if not self.adaptive:
            name = self.names[rng.randrange(len(self.names))]
        else:
            total = sum(self.w.values())
            r = rng.random() * total
            acc = 0.0
            name = self.names[-1]
            for n in self.names:
                acc += self.w[n]
                if r <= acc:
                    name = n
                    break
        self._uses[name] += 1
        self.total_uses[name] += 1
        return name

    def reward(self, name, kind):
        self._scores[name] += self.sigma[kind]

    def tick(self):
        self._iter += 1
        if self.adaptive and self._iter % self.segment == 0:
            for n in self.names:
                if self._uses[n] > 0:
                    self.w[n] = ((1.0 - self.reaction) * self.w[n]
                                 + self.reaction * (self._scores[n] / self._uses[n]))
                self._scores[n] = 0.0
                self._uses[n] = 0

    def stats(self):
        return {"weights": dict(self.w), "uses": dict(self.total_uses)}
```

```python
"""ALNS 본체. 항상-실행가능 불변식: 완성해만 수용/기록, 실패·거부는 snapshot 롤백."""
import random
import time
from dataclasses import dataclass
from functools import partial

from solver.alns.state import OpContext, SolutionState
from solver.alns import destroy as D
from solver.alns import repair as R
from solver.alns.accept import make_acceptance
from solver.alns.weights import OperatorSelector


@dataclass
class AlnsConfig:
    enabled: bool = True
    removal_rate: float = 0.33            # 설계서: 20~35%, 기본 0.33
    acceptance: str = "rrt"               # "rrt" | "sa"
    rrt_start_pct: float = 0.03           # 초기 임계 = 3% of initial obj (2~5% 범위)
    sa_start_pct: float = 0.05
    adaptive: bool = False                # 기본: 정적 균등
    segment: int = 100
    scores: tuple = (33, 13, 9)
    reaction: float = 0.1
    shaw_weights: tuple = (0.55, 1.36, 0.89)
    shaw_p: int = 6
    noise: float = 0.1
    disable_ops: tuple = ()               # 파괴/복구 공용 이름공간 (ablation용)
    max_iters: int | None = None          # 테스트/프로파일용 상한 (None=예산까지)
    seed: int = 0                         # shell이 improve rng 파생에 사용 (재현성)


def _build_ops(cfg):
    A, B, C = cfg.shaw_weights
    d = {
        "random": D.destroy_random,
        "worst_tardiness": D.destroy_worst_tardiness,
        "shaw": partial(D.destroy_shaw, A=A, B=B, C=C, p=cfg.shaw_p),
        "time_slice": D.destroy_time_slice,
        "spatial_column": D.destroy_spatial_column,
        "blocking_set": D.destroy_blocking_set,
    }
    r = {
        "greedy_noise": partial(R.repair_greedy_noise, noise=cfg.noise),
        "regret2": partial(R.repair_regret, k=2),
        "regret3": partial(R.repair_regret, k=3),
        "most_constrained": R.repair_most_constrained,
    }
    dsel = {k: v for k, v in d.items() if k not in cfg.disable_ops}
    rsel = {k: v for k, v in r.items() if k not in cfg.disable_ops}
    if not dsel or not rsel:
        raise ValueError(f"disable_ops={cfg.disable_ops}: 파괴/복구 연산자가 남지 않음")
    return dsel, rsel


def improve(core, incumbent, budget, *, prob, rng=None, cfg=None, trace=None, stats=None):
    cfg = cfg or AlnsConfig()
    if not cfg.enabled or not incumbent:
        return list(incumbent)
    rng = rng or random.Random(0)
    dops, rops = _build_ops(cfg)

    ctx = OpContext(prob)
    state = SolutionState(core, ctx, incumbent)
    n = len(incumbent)
    best = state.placements()
    best_obj = state.objective()
    cur_obj = best_obj
    acc = make_acceptance(cfg, best_obj)
    dsel = OperatorSelector(list(dops), cfg.adaptive, cfg.segment, cfg.scores, cfg.reaction)
    rsel = OperatorSelector(list(rops), cfg.adaptive, cfg.segment, cfg.scores, cfg.reaction)

    t0 = time.monotonic()
    total = max(budget.remain(), 1e-9)     # improve 진입 시점의 잔여 예산 = progress 분모
    it = 0
    if trace is not None:
        trace.append((0.0, best_obj))
    q = max(2, round(cfg.removal_rate * n))

    while not budget.expired() and (cfg.max_iters is None or it < cfg.max_iters):
        it += 1
        snap = state.snapshot()
        dname = dsel.select(rng)
        rname = rsel.select(rng)
        try:
            removed = dops[dname](state, q, rng, ctx)
            ok = rops[rname](state, removed, rng, ctx)
        except AssertionError:
            state.restore(snap)            # 방어: 어떤 연산자 이상도 incumbent 불변식 유지
            dsel.tick(); rsel.tick()
            continue
        if ok and len(state.placed) == n:
            cand_obj = state.objective()
            progress = min(1.0, (time.monotonic() - t0) / total)
            if cand_obj < best_obj - 1e-9:
                best, best_obj, cur_obj = state.placements(), cand_obj, cand_obj
                dsel.reward(dname, "best"); rsel.reward(rname, "best")
                if trace is not None:
                    trace.append((time.monotonic() - t0, best_obj))
            elif acc.accept(cand_obj, cur_obj, best_obj, progress, rng):
                kind = "better" if cand_obj < cur_obj - 1e-9 else "accepted"
                dsel.reward(dname, kind); rsel.reward(rname, kind)
                cur_obj = cand_obj
            else:
                state.restore(snap)        # 거부 → current 유지
        else:
            state.restore(snap)            # 복구 실패 → 롤백 (항상-실행가능)
        dsel.tick(); rsel.tick()

    if stats is not None:
        stats.update({"iters": it, "best_obj": best_obj,
                      "destroy": dsel.stats(), "repair": rsel.stats()})
    return best
```

```python
    from solver.alns.loop import improve as alns_improve, AlnsConfig
    improve.last_trace, improve.last_stats = [], {}          # 호출마다 초기화 (stale 방지)
    cfg = cfg or AlnsConfig()
    if not cfg.enabled or not incumbent:
        return incumbent
    trace, stats = [], {}
    try:
        out = alns_improve(core, incumbent, budget, prob=prob, rng=rng,
                           cfg=cfg, trace=trace, stats=stats)
    except Exception:
        return incumbent
    improve.last_trace, improve.last_stats = trace, stats    # solve가 report로 옮김
    return out if out else incumbent
```

`solve(prob_info, timelimit)`의 시그니처를 `solve(prob_info, timelimit, *, alns_cfg=None, report=None)`로 확장하고, 본문의 "멀티스타트 construct 루프" 이후를 다음 흐름으로 수정 (기존 trivial 안전망·최종 utils 검증·전역 try/except 골격은 유지):

```python
def _weighted_obj(core, prob_info):
    """core.objective() 튜플의 가중합. 1주차 shell에 동등 헬퍼가 이미 있으면 그것을 사용."""
    z1, z2, z3 = core.objective()
    w = prob_info.get("weights", {})
    return w.get("w1", 1.0) * z1 + w.get("w2", 1.0) * z2 + w.get("w3", 1.0) * z3
```

```python
def bench_earliest_feasible(core, ctx, state):
    ts = []
    for blk in sorted(state.placed)[:20]:
        p = state.remove_block(blk)
        t0 = time.perf_counter_ns()
        for _ in range(200):
            core.earliest_feasible(blk, p[1], p[4], ctx.release[blk], ctx.horizon)
```

```python
import json, sys

def main(construct_path, alns_path):
    c = json.load(open(construct_path))["results"]
    a = json.load(open(alns_path))["results"]
    names = sorted(c, key=lambda n: int(n.split("_")[1]))

    print(f"{'inst':>8} {'Z1(con)':>10} {'Z1(alns)':>10} {'ΔZ1%':>7} {'obj(con)':>12} {'obj(alns)':>12}")

        feas_c += bool(rc["feasible"]); feas_a += bool(ra["feasible"])
        sum_z1_c += rc["obj1"]; sum_z1_a += ra["obj1"]
        dz = 100.0 * (rc["obj1"] - ra["obj1"]) / rc["obj1"] if rc["obj1"] else 0.0
        print(f"{n:>8} {rc['obj1']:>10.1f} {ra['obj1']:>10.1f} {dz:>6.1f}% "
```

## 계약 변경 요청

1주차 계약(Core API·solver 모듈 시그니처)은 본 계획에서 **일절 수정하지 않는다**. 다만 아래 4건은 1주차 문서에 명시된 지점의 확장이므로 승인 후 1주차 계획서의 계약 절에도 동기화할 것.

1. **`shell.improve` 훅 키워드 인자 추가 (필수)** — 1주차: `improve(core, incumbent, budget)` pass-through. 요청: `improve(core, incumbent, budget, *, prob, rng, cfg=None) -> Placements`. 근거: Shaw 관련도(shape bbox), Z2 델타(workload), Z3(pref), 목적 가중치(w1/w2/w3)는 `prob_info`에만 있고 Core 계약에 노출되지 않음. 위치 인자 3개는 유지되므로 기존 문구와 충돌 없음. 호출자는 shell.solve 하나뿐.
2. **`shell.solve` 키워드 전용 인자 추가 (필수)** — `solve(prob_info, timelimit, *, alns_cfg=None, report=None)`. 근거: 벤치 ablation(--disable-op)과 수렴 trace 수집이 제출 경로(`myalgorithm.algorithm`은 여전히 `solve(prob_info, timelimit)` 위치 인자만 사용)와 분리된 채로 필요. 기본값이 None이라 1주차 호출부·테스트는 무변경.
3. **`ogc_core` 테스트 전용 바인딩 추가 (고지)** — `_nfp_bitmap(bi, oi, bj, oj, kind, method) -> (dx_lo, dy_lo, w, h, bytes)`. 1주차 `_overlap`/`_static_bit` 선례를 따르는 비계약(언더스코어) 바인딩으로, 공개 Core API 불변. 승인 불요라 판단하나 투명성 위해 기재.
4. **`Core.best_insertion` 추가 (조건부 — Task 11 결정 대기)** — Task 11 측정에서 Python 전용 시간 > 30%일 때만: `best_insertion(block, t_max) -> tuple[bay, x, y, orient, entry] | None` (전 bay×방향의 최소 (지각, y, x) 후보를 C++에서 일괄 탐색). 설계서 §4 L1의 "best_insertion — 디코더 내장 루프" 항목을 계약으로 승격하는 것. 트리거되지 않으면 요청 철회.
